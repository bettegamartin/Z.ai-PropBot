   106→def load_rth_raw(path):
   107→    df = pd.read_csv(path, index_col=0)
   108→    df.index = pd.to_datetime(df.index, utc=True).tz_convert(ET)
   109→    rth = df.between_time("10:00", "15:59")
   110→    return rth[rth["Volume"] > 0]
   111→
   112→def independent_signals(path, backtest_reentry_rule=True):
   113→    rth = load_rth_raw(path)
   114→    days = [g for _, g in rth.groupby(rth.index.normalize()) if len(g) >= 5][-5:]
   115→    trades, sig_bars = [], set()
   116→    for g in days:
   117→        tp = (g["High"] + g["Low"] + g["Close"]) / 3
   118→        vwap = (tp * g["Volume"]).cumsum() / g["Volume"].cumsum()
   119→        dev = g["Close"] - vwap
   120→        day_open = g["Open"].iloc[0]
   121→        pos = None
   122→        n_day = 0                                   # per-day cap, as the engine does
   123→        for i in range(2, len(g)):
   124→            if g.index[i].hour >= 14 or n_day >= 3:
   125→                break
   126→            if backtest_reentry_rule and pos is not None and i <= pos["exit_i"]:
   127→                continue
   128→            if abs(g["Close"].iloc[i] - day_open) > 0.006 * day_open:
   129→                continue
   130→            sd = dev.iloc[:i + 1].std()
   131→            if not np.isfinite(sd) or sd <= 0:
   132→                continue
   133→            up, dn = vwap.iloc[i] + 2.0 * sd, vwap.iloc[i] - 2.0 * sd
   134→            sig_bars.add(g.index[i].strftime("%H:%M"))
   135→            side = None
   136→            if g["High"].iloc[i] >= up:
   137→                side, ep, sp, tpx = -1, max(up, g["Open"].iloc[i]), None, vwap.iloc[i]
   138→                sp = ep + 1.2 * sd
   139→            elif g["Low"].iloc[i] <= dn:
   140→                side, ep, sp, tpx = 1, min(dn, g["Open"].iloc[i]), None, vwap.iloc[i]
   141→                sp = ep - 1.2 * sd
   142→            else:
   143→                continue
   144→            pos = dict(side=side, entry_px=ep, exit_i=None, exit_px=None, reason=None)
   145→            for j in range(i, len(g)):
   146→                hi, lo = g["High"].iloc[j], g["Low"].iloc[j]
   147→                if side > 0 and lo <= sp:
   148→                    pos.update(exit_px=sp, exit_i=j, reason="stop"); break
   149→                if side < 0 and hi >= sp:
   150→                    pos.update(exit_px=sp, exit_i=j, reason="stop"); break
   151→                if side > 0 and hi >= tpx + 0.25:
   152→                    pos.update(exit_px=tpx, exit_i=j, reason="tp"); break
   153→                if side < 0 and lo <= tpx - 0.25:
   154→                    pos.update(exit_px=tpx, exit_i=j, reason="tp"); break
   155→            if pos["exit_px"] is None:
   156→                pos.update(exit_px=g["Close"].iloc[len(g)-1], exit_i=len(g)-1, reason="eod")
   157→            pos.update(entry_ts=g.index[i], exit_ts=g.index[pos["exit_i"]])
   158→            trades.append(pos); n_day += 1
   159→    return trades, sig_bars

   270→def _day_trades(g, allow_same_bar_reentry):
   271→    tp = (g["High"] + g["Low"] + g["Close"]) / 3
   272→    vwap = (tp * g["Volume"]).cumsum() / g["Volume"].cumsum()
   273→    dev = g["Close"] - vwap
   274→    day_open = g["Open"].iloc[0]
   275→    trades, pos, n_day = [], None, 0
   276→    for i in range(2, len(g)):
   277→        if g.index[i].hour >= 14 or n_day >= 3:
   278→            break

   280→            continue

   282→            continue

   284→        if not np.isfinite(sd) or sd <= 0:
   285→            continue
   286→        up, dn = vwap.iloc[i] + 2 * sd, vwap.iloc[i] - 2 * sd
   287→        if g["High"].iloc[i] >= up:
   288→            side, ep, sp, tpx = -1, max(up, g["Open"].iloc[i]), None, vwap.iloc[i]
   289→        elif g["Low"].iloc[i] <= dn:
   290→            side, ep, sp, tpx = 1, min(dn, g["Open"].iloc[i]), None, vwap.iloc[i]
   291→        else:
   292→            continue
   293→        sp = ep + (1 if side < 0 else -1) * 1.2 * sd
   294→        pos = dict(side=side, entry_px=ep, exit_i=None, exit_px=None, reason=None)
   295→        for j in range(i, len(g)):
   296→            hi, lo = g["High"].iloc[j], g["Low"].iloc[j]
   297→            if side > 0 and lo <= sp:
   298→                pos.update(exit_px=sp, exit_i=j, reason="stop"); break

   300→                pos.update(exit_px=sp, exit_i=j, reason="stop"); break

   464→# TEST I — duplicate scheduled execution (concurrency model)
   465→# =====================================================================
   466→def snap():
   467→    for f in (ps.STATE_F, ps.TRADES_F):
   468→        if os.path.exists(f):
   469→            shutil.copy(f, f + ".snap")
   470→        else:
   471→            if os.path.exists(f + ".snap"):
   472→                os.remove(f + ".snap")
   473→
   474→def restore(state_only=False):
   475→    files = (ps.STATE_F,) if state_only else (ps.STATE_F, ps.TRADES_F)
   476→    for f in files:
   477→        if os.path.exists(f + ".snap"):
   478→            shutil.copy(f + ".snap", f)
   479→        elif os.path.exists(f):
   480→            os.remove(f)
   481→
   482→def run_once(bars):
   483→    st = json.load(open(ps.STATE_F)) if os.path.exists(ps.STATE_F) else ps.fresh_state()
   484→    done = set(st["done_bars"].get("ES", []))
   485→    for b in bars:
   486→        key = b[5].isoformat()
   487→        if key in done:
   488→            continue
   489→        ps.bar_step("ES", b, st, "live", now_ts=None)
   490→        done.add(key)
   491→    st["done_bars"]["ES"] = sorted(done)[-12:]
   492→    ps.atomic_json(ps.STATE_F, st)
   493→
