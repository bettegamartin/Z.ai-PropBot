   149→   149→def read_journal(path):
   150→   150→    """Return (set_of_trade_ids, n_valid_rows); repairs a truncated tail line."""
   151→   151→    if not os.path.exists(path):
   152→   152→        return set(), 0
   153→   153→    with open(path) as fh:
   154→   154→        lines = fh.read().splitlines()
   155→   155→    if not lines:
   156→   156→        return set(), 0
   157→   157→    header = lines[0].split(",")
   158→   158→    try:
   159→   159→        tid_i = header.index("trade_id")
   160→   160→    except ValueError:
   161→   161→        tid_i = None
   162→   162→    good, ids, dropped = [], set(), 0
   163→   163→    for ln in lines[1:]:
   164→   164→        parts = ln.split(",")
   165→   165→        if len(parts) != len(header):
   166→   166→            dropped += 1
   167→   167→            continue
   168→   168→        good.append(ln)
   169→   169→        if tid_i is not None:
   170→   170→            ids.add(parts[tid_i])
   171→   171→    if dropped:
   172→   172→        tmp = path + ".tmp"
   173→   173→        with open(tmp, "w") as fh:
   174→   174→            fh.write("\n".join([lines[0]] + good) + "\n")
   175→   175→        os.replace(tmp, path)
   176→   176→        print(f"JOURNAL REPAIR: dropped {dropped} malformed row(s) from {os.path.basename(path)}")
   177→   177→    return ids, len(good)

   180→   180→def flush_journal_rows(path, rows):
   181→   181→    """Append journal rows idempotently: a trade_id already on disk is skipped."""
   182→   182→    os.makedirs(os.path.dirname(path), exist_ok=True)
   183→   183→    ids, _ = read_journal(path)
   184→   184→    new = not os.path.exists(path)
   185→   185→    appended = skipped = 0
   186→   186→    with open(path, "a") as fh:
   187→   187→        if new:
   188→   188→            fh.write(",".join(JOURNAL_COLS) + "\n")
   189→   189→        for r in rows:
   190→   190→            tid = r.get("trade_id")
   191→   191→            if tid and tid in ids:
   192→   192→                skipped += 1
   193→   193→                continue
   194→   194→            fh.write(",".join(str(r.get(c, "")) for c in JOURNAL_COLS) + "\n")
   195→   195→            if tid:
   196→   196→                ids.add(tid)
   197→   197→            appended += 1
   198→   198→    return appended, skipped

   233→   233→    if skipped:
   234→   234→        print(f"JOURNAL DEDUP: {skipped} row(s) skipped (trade_id already on disk)")
   235→   235→    PENDING_ROWS.clear()
   236→   236→    PENDING_EQ.clear()
   237→   237→    PENDING_LOGS.clear()

   244→   244→def fetch_1h(yh, want_quote=False):
   245→   245→    """Yahoo 1h bars -> (DataFrame(O,H,L,C,V) tz ET, quote) for RTH bars.
   246→   246→
   247→   247→    Returns completed bars only (bar close + 90s <= now). When want_quote is
   248→   248→    True it also returns an observable-price dict for the forming bar:
   249→   249→      dict(px=latest price, ts=bar stamp, method='poll_quote') or None.
   250→   250→    Yahoo futures data is delayed/indicative -- the delay is NOT compensated
   251→   251→    anywhere and fill_method tags record which price class was used.
   252→   252→    """
   253→   253→    for host in ("query1", "query2"):
   254→   254→        url = (f"https://{host}.finance.yahoo.com/v8/finance/chart/{yh}"
   255→   255→               f"?range=5d&interval=1h")
   256→   256→        try:
   257→   257→            req = urllib.request.Request(url, headers={"User-Agent": UA})
   258→   258→            with urllib.request.urlopen(req, timeout=25) as r:
   259→   259→                js = json.load(r)
   260→   260→            res = js["chart"]["result"][0]
   261→   261→            ts = res.get("timestamp") or []
   262→   262→            q = res["indicators"]["quote"][0]
   263→   263→            now = datetime.now(ET)
   264→   264→            rows, forming = [], None
   265→   265→            for i, t in enumerate(ts):
   266→   266→                o, h, l, c, v = (q[k][i] for k in ("open", "high", "low", "close", "volume"))
   267→   267→                if None in (o, h, l, c) or not v:
   268→   268→                    continue
   269→   269→                dt = datetime.fromtimestamp(t, tz=ZoneInfo("UTC")).astimezone(ET)
   270→   270→                rec = (dt, float(o), float(h), float(l), float(c), float(v))
   271→   271→                rows.append(rec)
   272→   272→                if dt + timedelta(hours=1) > now - timedelta(seconds=90):
   273→   273→                    forming = rec                     # most recent still-forming bar
   274→   274→            if not rows:
   275→   275→                continue
   276→   276→            df = pd.DataFrame(rows, columns=["ts", "Open", "High", "Low", "Close", "Volume"])
   277→   277→            df = df.set_index("ts").sort_index()
   278→   278→            df = df.between_time("10:00", "15:59")          # clean RTH bars
   279→   279→            df = df[[t + timedelta(hours=1) <= now - timedelta(seconds=90)
   280→   280→                     for t in df.index]]                     # completed bars only
   281→   281→            quote = None
   282→   282→            if want_quote and forming is not None:
   283→   283→                quote = dict(px=forming[4], ts=forming[0], method="poll_quote")
   284→   284→                if len(df) and abs(quote["px"] - float(df["Close"].iloc[-1])) > QUOTE_SANE_PCT * float(df["Close"].iloc[-1]):
   285→   285→                    quote = None                            # implausible forming price -> distrust
   286→   286→            return (df, quote) if want_quote else df
   287→   287→        except Exception:
   288→   288→            continue
   289→   289→    return (None, None) if want_quote else None

   376→   376→def check_daily_stop(st, d, sym):
   377→   377→    """P0.1: account-level daily loss stop (Table 8: halt for the day at -$400).
   378→   378→
   379→   379→    Evaluated on realized day P&L across BOTH symbols after every close; sets
   380→   380→    halted_date in state (persists across restarts) and per-symbol halted flags.
   381→   381→    Resets automatically when a new trading day starts (halted_date is compared
   382→   382→    against the bar date everywhere).
   383→   383→    """
   384→   384→    if st.get("halted_date") == d["date"]:
   385→   385→        return
   386→   386→    day = account_day_pnl(st, d["date"])
   387→   387→    if day <= -P["dstop"]:
   388→   388→        st["halted_date"] = d["date"]
   389→   389→        st["halted_reason"] = (f"account day P&L {day:+.2f} breached the "
   390→   390→                               f"-${P['dstop']:.0f} daily stop")
   391→   391→        for dd in st["day"].values():
   392→   392→            if dd and dd["date"] == d["date"]:
   393→   393→                dd["halted"] = True
   394→   394→                dd["halted_reason"] = st["halted_reason"]
   395→   395→        buf_log(f"**DAILY STOP {d['date']}: {st['halted_reason']} — no further entries "
   396→   396→                f"on any symbol until the next trading day (sym {sym} closed a trade)**")
   397→   397→

   399→   399→def close_pos(d, px, ts, reason, sym, st, mode):
   400→   400→    pos = d["pos"]
   401→   401→    if not isinstance(ts, datetime):
   402→   402→        ts = datetime.fromisoformat(str(ts))
   403→   403→    ets = pos["entry_ts"] if isinstance(pos["entry_ts"], datetime) \
   404→   404→        else datetime.fromisoformat(str(pos["entry_ts"]))
   405→   405→    spec = SPEC[sym]
   406→   406→    per_ct = (px - pos["entry_px"]) * pos["side"] * spec["dpp"]
   407→   407→    costs = (spec["comm"] * 2 + spec["slip"] * spec["tick_usd"] * 2) * pos["contracts"]
   408→   408→    net = per_ct * pos["contracts"] - costs
   409→   409→    d["day_realized"] += net
   410→   410→    st["cum"] += net
   411→   411→    st["n_trades"] += 1
   412→   412→    PENDING_ROWS.append(dict(
   413→   413→        mode=mode, trade_id=pos.get("trade_id", ""), ts=ts.isoformat(), sym=sym,
   414→   414→        side="long" if pos["side"] > 0 else "short",
   415→   415→        contracts=pos["contracts"],
   416→   416→        signal_px=round(pos.get("signal_px", pos["entry_px"]), 2),
   417→   417→        signal_ts=pos.get("signal_ts", ets).isoformat() if isinstance(pos.get("signal_ts", ets), datetime)
   418→   418→                  else str(pos.get("signal_ts", ets)),
   419→   419→        entry_px=round(pos["entry_px"], 2),
   420→   420→        entry_ts=ets.isoformat(), exit_ts=ts.isoformat(),
   421→   421→        exit_px=round(px, 2), reason=reason,
   422→   422→        fill_method=pos.get("fill_method", "bar_fill"),
   423→   423→        exec_lag_min=pos.get("exec_lag_min", 0),
   424→   424→        gross_usd=round(per_ct * pos["contracts"], 2),
   425→   425→        costs_usd=round(costs, 2), net_usd=round(net, 2), cum_eq=round(st["cum"], 2)))
   426→   426→    buf_log(f"`{mode}` {ts:%m-%d %H:%M} ET **{sym} {pos['contracts']} micro "
   427→   427→            f"{'LONG' if pos['side'] > 0 else 'SHORT'} closed ({reason})** "
   428→   428→            f"{pos['entry_px']:.2f} -> {px:.2f} = ${net:+.2f} (day {d['day_realized']:+.2f}, cum ${st['cum']:+.2f})")
   429→   429→    d["pos"] = None
   430→   430→    check_daily_stop(st, d, sym)
   431→   431→    return net

   442→   442→def live_enter(sym, st, d, side, signal_px, sig_bar, sd, vwap, now_ts, quote, mode):
   443→   443→    """P0.2: enter at the first price observable at action time.
   444→   444→
   445→   445→    signal_px  = band price the researched strategy would have filled at the
   446→   446→                 signal bar (recorded only, NEVER used as a live fill).
   447→   447→    fill price = observable price carried in `quote` (poll forming price, or
   448→   448→                 the newest completed close when no forming bar exists and the
   449→   449→                 completed close is <= 40 min old).
   450→   450→    Stop/target are recomputed from the fill; target stays the signal-bar VWAP.
   451→   451→    """
   452→   452→    ts = sig_bar[5]
   453→   453→    fill_px = float(quote["px"])
   454→   454→    method = quote.get("method", "poll_quote")
   455→   455→    lag = max(int((now_ts - (ts + timedelta(hours=1))).total_seconds() // 60), 0)
   456→   456→    tid = f"{ts.date().isoformat()}_{sym}_{d['trades'] + 1:02d}"
   457→   457→    if side < 0:
   458→   458→        pos = dict(side=-1, entry_px=fill_px, stop_px=fill_px + P["stop_k"] * sd,
   459→   459→                   tp_px=vwap, signal_px=signal_px, signal_ts=ts,
   460→   460→                   entry_ts=now_ts, entry_exec_ts=now_ts, contracts=contracts_for(st),
   461→   461→                   fill_method=method, exec_lag_min=lag, trade_id=tid)
   462→   462→    else:
   463→   463→        pos = dict(side=1, entry_px=fill_px, stop_px=fill_px - P["stop_k"] * sd,
   464→   464→                   tp_px=vwap, signal_px=signal_px, signal_ts=ts,
   465→   465→                   entry_ts=now_ts, entry_exec_ts=now_ts, contracts=contracts_for(st),
   466→   466→                   fill_method=method, exec_lag_min=lag, trade_id=tid)
   467→   467→    d["pos"] = pos
   468→   468→    d["trades"] += 1
   469→   469→    st["n_entries"] += 1
   470→   470→    buf_log(f"`{mode}` {now_ts:%m-%d %H:%M} ET **{sym} {pos['contracts']} micro "
   471→   471→            f"{'SHORT' if side < 0 else 'LONG'} opened** signal {signal_px:.2f} "
   472→   472→            f"({ts:%H:%M} bar) -> filled {fill_px:.2f} @ {now_ts:%H:%M} "
   473→   473→            f"({method}, lag {lag}m; stop {pos['stop_px']:.2f}, target {pos['tp_px']:.2f}, "
   474→   474→            f"sd {sd:.2f}, id {tid})")
   475→   475→    # target already traded through at action time -> exit at the observable price
   476→   476→    tick = 0.25
   477→   477→    if side < 0 and fill_px <= pos["tp_px"] - tick:
   478→   478→        buf_log(f"`{mode}` target already traded through at action time "
   479→   479→                f"({sym} fill {fill_px:.2f} <= target-tick {pos['tp_px'] - tick:.2f}) "
   480→   480→                f"-> immediate exit at the observable price")
   481→   481→        close_pos(d, fill_px, now_ts, "tp_poll", sym, st, mode)
   482→   482→    elif side > 0 and fill_px >= pos["tp_px"] + tick:
   483→   483→        buf_log(f"`{mode}` target already traded through at action time "
   484→   484→                f"({sym} fill {fill_px:.2f} >= target+tick {pos['tp_px'] + tick:.2f}) "
   485→   485→                f"-> immediate exit at the observable price")
   486→   486→        close_pos(d, fill_px, now_ts, "tp_poll", sym, st, mode)
   487→   487→    return pos

   490→   490→def bar_step(sym, bar, st, mode, now_ts=None, quote=None):
   491→   491→    o, h, l, c, v, ts = bar
   492→   492→    if mode == "live" and now_ts is None:
   493→   493→        now_ts = datetime.now(ET)          # defensive: live entries need a wall clock
   494→   494→    spec = SPEC[sym]
   495→   495→    if sym not in st["day"] or st["day"][sym]["date"] != ts.date().isoformat():
   496→   496→        prev = st["day"].get(sym)
   497→   497→        if prev and prev.get("pos"):
   498→   498→            close_pos(prev, prev["last_close"], prev["last_ts"], "eod_missed", sym, st, mode)
   499→   499→        st["day"][sym] = new_day(ts.date().isoformat())
   500→   500→    d = st["day"][sym]
   501→   501→    if d["day_open"] is None:
   502→   502→        d["day_open"] = o
   503→   503→    # session VWAP + expanding sigma (includes current bar, as in backtest)
   504→   504→    d["vwap_n"] += (h + l + c) / 3 * v
   505→   505→    d["vwap_d"] += v
   506→   506→    vwap = d["vwap_n"] / max(d["vwap_d"], 1e-9)
   507→   507→    devs = d["devs"] + [c - vwap]
   508→   508→    sd = float(np.std(devs, ddof=1)) if len(devs) >= 2 else 0.0
   509→   509→    d["last_close"] = c
   510→   510→    d["last_ts"] = ts
   511→   511→    tick = 0.25
   512→   512→    live = (mode == "live")
   513→   513→
   514→   514→    # 1) exits on an open position: stop first, then target (trade-through).
   515→   515→    #    In live mode a resting order could not have existed before the entry
   516→   516→    #    was actioned, so bars that closed before the entry action are skipped.
   517→   517→    if open_pos(d):
   518→   518→        pos = d["pos"]
   519→   519→        exec_ts = _as_dt(pos["entry_exec_ts"]) if pos.get("entry_exec_ts") else None
   520→   520→        predates = live and exec_ts is not None and (ts + timedelta(hours=1)) <= exec_ts
   521→   521→        if not predates:
   522→   522→            if pos["side"] > 0 and l <= pos["stop_px"]:
   523→   523→                close_pos(d, pos["stop_px"], ts, "stop", sym, st, mode)
   524→   524→            elif pos["side"] < 0 and h >= pos["stop_px"]:
   525→   525→                close_pos(d, pos["stop_px"], ts, "stop", sym, st, mode)
   526→   526→            elif pos["side"] > 0 and h >= pos["tp_px"] + tick:
   527→   527→                close_pos(d, pos["tp_px"], ts, "tp", sym, st, mode)
   528→   528→            elif pos["side"] < 0 and l <= pos["tp_px"] - tick:
   529→   529→                close_pos(d, pos["tp_px"], ts, "tp", sym, st, mode)
   530→   530→
   531→   531→    # 2) entries: bars stamped 12:00/13:00 ET (i>=2, hour<14), fresh bar only.
   532→   532→    #    P0.1: the daily-stop gate checks BOTH the persisted halted_date flag
   533→   533→    #    and the live account-day P&L condition, so a restart (or any path
   534→   534→    #    that missed the flag) still cannot enter past the -$400 stop.
   535→   535→    if (not open_pos(d) and len(devs) >= 3 and ts.hour < P["cutoff_h"]
   536→   536→            and d["trades"] < P["max_trades"] and not d["halted"]
   537→   537→            and st.get("halted_date") != ts.date().isoformat()
   538→   538→            and account_day_pnl(st, ts.date().isoformat()) > -P["dstop"]):
   539→   539→        fresh = now_ts is None or (ts + timedelta(hours=1)) >= now_ts - timedelta(minutes=FRESH_MIN)
   540→   540→        if fresh and abs(c - d["day_open"]) <= P["trend"] * d["day_open"] and sd > 0:
   541→   541→            up, dn = vwap + P["k"] * sd, vwap - P["k"] * sd
   542→   542→            if h >= up:
   543→   543→                side, signal_px = -1, max(up, o)
   544→   544→            elif l <= dn:
   545→   545→                side, signal_px = 1, min(dn, o)
   546→   546→            else:
   547→   547→                side, signal_px = None, None
   548→   548→            if side is not None:
   549→   549→                if live:
   550→   550→                    if quote is None:
   551→   551→                        buf_log(f"`{mode}` {ts:%m-%d %H:%M} ET {sym} signal "
   552→   552→                                f"({'SHORT' if side < 0 else 'LONG'} @ {signal_px:.2f}) NOT taken: "
   553→   553→                                f"no fresh observable price at the acting poll (stale/delayed feed)")
   554→   554→                    else:
   555→   555→                        live_enter(sym, st, d, side, signal_px, bar, sd, vwap, now_ts, quote, mode)
   556→   556→                else:
   557→   557→                    # replay/signal mode: backtest bar-level fill + same-bar exits
   558→   558→                    n = contracts_for(st)
   559→   559→                    tid = f"{ts.date().isoformat()}_{sym}_{d['trades'] + 1:02d}"
   560→   560→                    if side < 0:
   561→   561→                        pos = dict(side=-1, entry_px=signal_px, stop_px=signal_px + P["stop_k"] * sd,
   562→   562→                                   tp_px=vwap, signal_px=signal_px, signal_ts=ts,
   563→   563→                                   entry_ts=ts, contracts=n, fill_method="bar_fill",
   564→   564→                                   exec_lag_min=0, trade_id=tid)
   565→   565→                    else:
   566→   566→                        pos = dict(side=1, entry_px=signal_px, stop_px=signal_px - P["stop_k"] * sd,
   567→   567→                                   tp_px=vwap, signal_px=signal_px, signal_ts=ts,
   568→   568→                                   entry_ts=ts, contracts=n, fill_method="bar_fill",
   569→   569→                                   exec_lag_min=0, trade_id=tid)
   570→   570→                    d["pos"] = pos
   571→   571→                    d["trades"] += 1
   572→   572→                    st["n_entries"] += 1
   573→   573→                    buf_log(f"`{mode}` {ts:%m-%d %H:%M} ET **{sym} {n} micro "
   574→   574→                            f"{'SHORT' if pos['side'] < 0 else 'LONG'} opened** @ {pos['entry_px']:.2f} "
   575→   575→                            f"(stop {pos['stop_px']:.2f}, target {pos['tp_px']:.2f}, sd {sd:.2f}, id {tid})")
   576→   576→                    # same-bar stop/target (replicates backtest executor)
   577→   577→                    if pos["side"] < 0 and h >= pos["stop_px"]:
   578→   578→                        close_pos(d, pos["stop_px"], ts, "stop", sym, st, mode)
   579→   579→                    elif pos["side"] > 0 and l <= pos["stop_px"]:
   580→   580→                        close_pos(d, pos["stop_px"], ts, "stop", sym, st, mode)
   581→   581→                    elif pos["side"] < 0 and l <= pos["tp_px"] - tick:
   582→   582→                        close_pos(d, pos["tp_px"], ts, "tp", sym, st, mode)
   583→   583→                    elif pos["side"] > 0 and h >= pos["tp_px"] + tick:
   584→   584→                        close_pos(d, pos["tp_px"], ts, "tp", sym, st, mode)
   585→   585→
   586→   586→    # 3) EOD flat on the 15:00-16:00 bar (or any late bar at day end)
   587→   587→    if open_pos(d) and ts.hour == 15:
   588→   588→        close_pos(d, c, ts, "eod", sym, st, mode)
   589→   589→
   590→   590→    d["devs"] = devs
   591→   591→    return d

   594→   594→def open_pnl(st):
   595→   595→    tot = 0.0
   596→   596→    for sym, d in st["day"].items():
   597→   597→        if d and d.get("pos") and (d.get("mark_px") or d.get("last_close")) is not None:
   598→   598→            spec = SPEC[sym]
   599→   599→            pos = d["pos"]
   600→   600→            px = d.get("mark_px", d["last_close"])
   601→   601→            tot += (px - pos["entry_px"]) * pos["side"] * spec["dpp"] * pos["contracts"]
   602→   602→    return tot

   605→   605→def overlay_update(st, eod=False, write=True):
   606→   606→    """Overlay accounting per research Table 4.
   607→   607→
   608→   608→    Both floors are EOD trails: hwm_eod is the high-water mark of CLOSED equity
   609→   609→    at day summaries; intraday equity never raises the floor. Topstep pass
   610→   610→    additionally requires the best day < 55% of target (consistency). Bust is
   611→   611→    still evaluated against current equity (a live breach would be real).
   612→   612→    """
   613→   613→    now = datetime.now(ET)
   614→   614→    eq = st["cum"] + open_pnl(st)
   615→   615→    day_pnl = sum(d["day_realized"] for d in st["day"].values() if d)
   616→   616→    msgs = []
   617→   617→    for key, ov in st["overlays"].items():
   618→   618→        cfg = OVERLAYS[key]
   619→   619→        if ov.get("done"):
   620→   620→            continue
   621→   621→        oeq = eq - ov["base_cum"]
   622→   622→        # EOD bookkeeping first, so the floor derived below reflects this EOD
   623→   623→        if eod and st["traded_today"]:
   624→   624→            ov["tdays"] += 1
   625→   625→            ov["hwm_eod"] = max(ov["hwm_eod"], st["cum"] - ov["base_cum"])
   626→   626→            ov["best_day"] = max(ov.get("best_day", 0.0), day_pnl)
   627→   627→        # EOD trail for BOTH overlays (P1.3; the old intraday Apex trail is gone)
   628→   628→        floor = ov["hwm_eod"] - cfg["dd"]
   629→   629→        if cfg.get("floor_cap_be"):
   630→   630→            floor = min(floor, 0.0)            # Topstep: floor never rises above breakeven
   631→   631→        ov["floor"] = floor
   632→   632→        if oeq <= ov["floor"]:
   633→   633→            ov["fees"] += cfg["eval_fee"]
   634→   634→            ov["attempts"] += 1
   635→   635→            if write:
   636→   636→                buf_log(f"**{cfg['name']} overlay: WOULD BE BUSTED** {now:%m-%d %H:%M} ET "
   637→   637→                        f"(eq {oeq:+.0f} vs floor {ov['floor']:+.0f}). New attempt, "
   638→   638→                        f"fees so far ${ov['fees']:.0f}, attempts {ov['attempts']}")
   639→   639→            ov.update(base_cum=st["cum"], hwm_eod=0.0, floor=-cfg["dd"],
   640→   640→                      tdays=0, fees=ov["fees"], attempts=ov["attempts"],
   641→   641→                      best_day=0.0, start=now.date().isoformat())
   642→   642→            continue
   643→   643→        if day_pnl <= -P["dll"] and key == "topstep" and not ov.get("dll_today") == st["today"]:
   644→   644→            ov["dll_today"] = st["today"]
   645→   645→            msgs.append(f"Topstep DLL ${-P['dll']:.0f} hit (day {day_pnl:+.0f}) -> would flatten for the day")
   646→   646→        ocum = st["cum"] - ov["base_cum"]
   647→   647→        if ocum >= TARGET and ov["tdays"] >= cfg["min_days"]:
   648→   648→            cap = cfg.get("consistency")
   649→   649→            cap_usd = round(cap * TARGET, 2) if cap is not None else None
   650→   650→            if cap is not None and ov.get("best_day", 0.0) >= cap_usd:
   651→   651→                msg = (f"{cfg['name']} target reached but the consistency cap blocks a PASS "
   652→   652→                       f"(best day ${ov.get('best_day', 0.0):.0f} >= {cap_usd:.0f} = "
   653→   653→                       f"{cap:.0%} of target) -- keeps trading")
   654→   654→                if msg not in msgs:
   655→   655→                    msgs.append(msg)
   656→   656→                continue
   657→   657→            ov["fees"] += cfg["act"]
   658→   658→            ov["done"] = True
   659→   659→            ov["passed_on"] = now.date().isoformat()
   660→   660→            if write:
   661→   661→                buf_log(f"**{cfg['name']} overlay: WOULD HAVE PASSED the evaluation** on "
   662→   662→                        f"{ov['passed_on']} (profit {ocum:+.0f}, {ov['tdays']} trading days, "
   663→   663→                        f"best day ${ov.get('best_day', 0.0):.0f}, "
   664→   664→                        f"eval fees paid ${ov['fees']:.0f}, attempts {ov['attempts']})")
   665→   665→    return eq, msgs

   668→   668→def eod_summary(st):
   669→   669→    if st.get("last_summary") == st["today"]:
   670→   670→        return None
   671→   671→    st["last_summary"] = st["today"]
   672→   672→    day_pnl = sum(d["day_realized"] for d in st["day"].values() if d)
   673→   673→    n_today = sum(d["trades"] for d in st["day"].values() if d)
   674→   674→    if st["traded_today"]:
   675→   675→        st["trading_days"] += 1
   676→   676→    eq, msgs = overlay_update(st, eod=True)
   677→   677→    lines = [f"**DAY SUMMARY {st['today']}** — trades: {n_today}, day P&L: "
   678→   678→             f"${day_pnl:+.2f}, cumulative: ${st['cum']:+.2f} "
   679→   679→             f"({st['cum'] / TARGET * 100:+.0f}% of $3,000 target)"]
   680→   680→    if st.get("halted_date") == st["today"]:
   681→   681→        lines.append(f"DAILY STOP active today: {st.get('halted_reason', '')}")
   682→   682→    for key, ov in st["overlays"].items():
   683→   683→        cfg = OVERLAYS[key]
   684→   684→        if ov.get("done"):
   685→   685→            lines.append(f"{cfg['name']}: PASSED {ov['passed_on']} (fees ${ov['fees']:.0f})")
   686→   686→        else:
   687→   687→            ocum = st["cum"] - ov["base_cum"]
   688→   688→            lines.append(f"{cfg['name']}: profit {ocum:+.0f}/{TARGET:.0f}, floor "
   689→   689→                         f"{ov['floor']:+.0f}, tdays {ov['tdays']}, best day "
   690→   690→                         f"${ov.get('best_day', 0.0):+.0f}, fees ${ov['fees']:.0f}, "
   691→   691→                         f"attempts {ov['attempts']}")
   692→   692→    for m in msgs:
   693→   693→        lines.append(m)
   694→   694→    with open(LOG_F, "a") as fh:
   695→   695→        fh.write("\n".join(x if x.startswith("**") else f"- {x}" for x in lines) + "\n")
   696→   696→    PENDING_EQ.append(dict(ts=datetime.now(ET).isoformat(), eq=round(st["cum"], 2),
   697→   697→                           day=round(day_pnl, 2)))
   698→   698→    return " ".join(lines)

   707→   707→def classify_feed(df, quote, now):
   708→   708→    """Feed health for one symbol: never reports stale data as healthy."""
   709→   709→    if df is None or df.empty:
   710→   710→        return "unavailable"
   711→   711→    has_today = any(ts.date() == now.date() for ts in df.index)
   712→   712→    newest_close = df.index[-1] + timedelta(hours=1)
   713→   713→    lag = (now - newest_close).total_seconds() / 60.0
   714→   714→    if quote is not None:
   715→   715→        return "ok"
   716→   716→    if has_today and lag > STALE_LAG_MIN:
   717→   717→        return f"stale (newest completed bar {lag:.0f} min old)"
   718→   718→    if not has_today and in_rth(now) and now.hour >= 11:
   719→   719→        return f"stale (no completed bars today as of {now:%H:%M} ET)"
   720→   720→    if not has_today:
   721→   721→        return "awaiting first completed bar"
   722→   722→    return "ok"

   726→   726→    os.makedirs(OUT, exist_ok=True)

   734→   734→        pending_clear()
   735→   735→        now = now or datetime.now(ET)
   736→   736→        fetch = fetch or FETCH_IMPL
   737→   737→        st = load_state()
   738→   738→        st["today"] = now.date().isoformat()
   739→   739→        st["traded_today"] = False
   740→   740→        logline_now(f"--- live run {now.isoformat()} ---")
   741→   741→        if now.weekday() >= 5:
   742→   742→            print(f"WEEKEND {now:%a %H:%M} ET — market closed, no action.")
   743→   743→            return
   744→   744→        notes = []
   745→   745→        for sym in ("ES", "NQ"):
   746→   746→            try:
   747→   747→                res = fetch(SPEC[sym]["yh"], want_quote=True)
   748→   748→            except TypeError:
   749→   749→                res = fetch(SPEC[sym]["yh"])
   750→   750→            if isinstance(res, tuple):
   751→   751→                df, quote = res
   752→   752→            else:
   753→   753→                df, quote = res, None
   754→   754→            status = classify_feed(df, quote, now)
   755→   755→            st["feed_status"][sym] = status
   756→   756→            if status == "unavailable":
   757→   757→                notes.append(f"{sym}: feed unavailable")
   758→   758→                continue
   759→   759→            if status != "ok" and not status.startswith("awaiting"):
   760→   760→                notes.append(f"{sym}: feed {status}")
   761→   761→            if status.startswith("awaiting"):
   762→   762→                notes.append(f"{sym}: awaiting first completed bar")
   763→   763→            if quote is None and df is not None and len(df):
   764→   764→                # fallback observable: newest completed close if it is fresh enough
   765→   765→                newest_close = df.index[-1] + timedelta(hours=1)
   766→   766→                if (now - newest_close).total_seconds() / 60.0 <= FRESH_MIN:
   767→   767→                    quote = dict(px=float(df["Close"].iloc[-1]), ts=df.index[-1],
   768→   768→                                 method="last_completed_close")
   769→   769→            done = set(st["done_bars"].get(sym, []))
   770→   770→            rejected = set(st["rejected_bars"].get(sym, []))
   771→   771→            last_ts = st["last_bar_ts"].get(sym)
   772→   772→            rej_new = 0
   773→   773→            for ts, row in df.iterrows():
   774→   774→                key = ts.isoformat()
   775→   775→                if key in done:
   776→   776→                    continue
   777→   777→                if last_ts and ts <= datetime.fromisoformat(last_ts):
   778→   778→                    if key not in rejected:
   779→   779→                        rejected.add(key)
   780→   780→                        rej_new += 1
   781→   781→                    continue          # P1.4: never rewind state with old bars
   782→   782→                d = bar_step(sym, (row["Open"], row["High"], row["Low"], row["Close"],
   783→   783→                                   row["Volume"], ts), st, "live", now_ts=now, quote=quote)
   784→   784→                done.add(key)
   785→   785→                st["last_bar_ts"][sym] = key
   786→   786→                if d["trades"]:
   787→   787→                    st["traded_today"] = True
   788→   788→            st["done_bars"][sym] = sorted(done)[-MAX_KEY_HIST:]
   789→   789→            st["rejected_bars"][sym] = sorted(rejected)[-MAX_KEY_HIST:]
   790→   790→            if rej_new:
   791→   791→                notes.append(f"{sym}: {rej_new} stale/out-of-order bar(s) rejected (no rewind)")
   792→   792→            dd = st["day"].get(sym)
   793→   793→            if dd and dd["date"] == st["today"] and dd.get("pos") and quote is not None:
   794→   794→                dd["mark_px"] = float(quote["px"])
   795→   795→        if not st["traded_today"]:
   796→   796→            st["traded_today"] = any(d["trades"] for d in st["day"].values() if d)
   797→   797→        eq, dllmsgs = overlay_update(st, eod=False)
   798→   798→        summary = None
   799→   799→        if now.hour >= 16:
   800→   800→            summary = eod_summary(st)
   801→   801→        flush_pending(st)
   802→   802→        pos_desc = []
   803→   803→        for sym, d in st["day"].items():
   804→   804→            if d and d.get("pos"):
   805→   805→                p = d["pos"]
   806→   806→                pos_desc.append(f"{sym} {'L' if p['side'] > 0 else 'S'}{p['contracts']}"
   807→   807→                                f"@{p['entry_px']:.2f}")
   808→   808→        ov = st["overlays"]
   809→   809→        ap, ts_ = ov["apex"], ov["topstep"]
   810→   810→        halt_mark = " | DAILY STOP ACTIVE" if st.get("halted_date") == st["today"] else ""
   811→   811→        fs = st.get("feed_status", {})
   812→   812→        short = lambda s: str(s).split()[0] if s else "?"   # ok / stale / awaiting / unavailable
   813→   813→        feed_note = f"feed ES:{short(fs.get('ES'))} NQ:{short(fs.get('NQ'))}"
   814→   814→        if notes:
   815→   815→            feed_note += " | " + " | ".join(notes)
   816→   816→        print(f"LIVE {now:%a %m-%d %H:%M} ET | cum ${st['cum']:+.2f} | open: "
   817→   817→              f"{', '.join(pos_desc) if pos_desc else 'flat'} | "
   818→   818→              f"Apex floor {ap['floor']:+.0f} | Topstep floor {ts_['floor']:+.0f} | "
   819→   819→              f"{feed_note}{halt_mark}")
   820→   820→        if summary:
   821→   821→            print(summary)
   822→   822→    finally:
   823→   823→        try:
   824→   824→            fcntl.flock(lock, fcntl.LOCK_UN)
   825→   825→        except OSError:
   826→   826→            pass
   827→   827→        lock.close()

   832→   832→def run_replay(n):
   833→   833→    """Replay runs in an ISOLATED journal/state space: <current home>/replay/.
   834→   834→
   835→   835→    P1.2: it can never write the live trades.csv/state.json/log.md, and it
   836→   836→    aborts outright if the resolved replay directory would be the live root.
   837→   837→    """
   838→   838→    rdir = replay_dir_for(OUT)
   839→   839→    if rdir == os.path.abspath(ROOT_HOME) or rdir == os.path.abspath(OUT):
   840→   840→        print("ABORT: replay directory resolves to the live directory -- refusing to run.")
   841→   841→        return
   842→   842→    prev = (OUT, STATE_F, TRADES_F, EQUITY_F, LOG_F, LOCK_F)
   843→   843→    os.makedirs(rdir, exist_ok=True)
   844→   844→    set_paths(rdir)
   845→   845→    pending_clear()
   846→   846→    try:
   847→   847→        st = fresh_state()
   848→   848→        st["today"] = "replay"
   849→   849→        print(f"REPLAY of last {n} stored RTH sessions through the paper engine "
   850→   850→              f"(2 micros/symbol, real costs; ISOLATED journal: {TRADES_F}):\n")
   851→   851→        day_list = replay_bars("ES", n)
   852→   852→        dates = [g.index[0].date() for g in day_list]
   853→   853→        for di, date in enumerate(dates):
   854→   854→            for sym in ("ES", "NQ"):
   855→   855→                days = replay_bars(sym, n)
   856→   856→                # guard: only replay an NQ session whose date truly matches (audit M7)
   857→   857→                g = days[di]
   858→   858→                if g.index[0].date().isoformat() != date.isoformat():
   859→   859→                    print(f"  {date} {sym}: no matching stored session -- skipped "
   860→   860→                          f"(positional misalignment guard)")
   861→   861→                    continue
   862→   862→                for ts, row in g.iterrows():
   863→   863→                    bar_step(sym, (row["Open"], row["High"], row["Low"], row["Close"],
   864→   864→                                   row["Volume"], ts), st, "replay")
   865→   865→            st["traded_today"] = any(dd["trades"] for dd in st["day"].values() if dd)
   866→   866→            overlay_update(st, eod=True)
   867→   867→            dp = sum(dd["day_realized"] for dd in st["day"].values() if dd)
   868→   868→            nt = sum(dd["trades"] for dd in st["day"].values() if dd)
   869→   869→            print(f"  {date}  trades {nt:2d}  day ${dp:+8.2f}  cum ${st['cum']:+9.2f}  "
   870→   870→                  f"({st['cum'] / TARGET * 100:+5.0f}% of target)")
   871→   871→            st["day"] = {}
   872→   872→        flush_pending(st)
   873→   873→        ap, ts_ = st["overlays"]["apex"], st["overlays"]["topstep"]
   874→   874→        print(f"\n  replay totals: {st['n_trades']} closed trades over {n} sessions, "
   875→   875→              f"cum ${st['cum']:+.2f}")
   876→   876→        print(f"  Apex overlay: profit {st['cum'] - ap['base_cum']:+.0f}, floor {ap['floor']:+.0f}, "
   877→   877→              f"tdays {ap['tdays']}, busted {ap['attempts'] - 1}x")
   878→   878→        print(f"  Topstep overlay: profit {st['cum'] - ts_['base_cum']:+.0f}, floor {ts_['floor']:+.0f}, "
   879→   879→              f"tdays {ts_['tdays']}, busted {ts_['attempts'] - 1}x")
   880→   880→        print(f"\n(replay wrote ONLY to {rdir}; live journal/state untouched)")
   881→   881→    finally:
   882→   882→        set_paths(prev[0])
   883→   883→        pending_clear()
